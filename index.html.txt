function baslat() {
  adi = document.getElementById("inp-adi").value.trim();
  no  = document.getElementById("inp-no").value.trim();
  if (!adi || !no) { alert("Lütfen adınızı ve öğrenci numaranızı giriniz."); return; }

  // Seed = öğrenci no + hafta → herkese farklı, aynı öğrenciye aynı hafta hep aynı
  seed = 0;
  for (let c of (no + HAFTA_NO)) seed = (seed * 31 + c.charCodeAt(0)) | 0;
  seed = Math.abs(seed);

  sorular = soruUret(seed);
  goster("sinav");
  document.getElementById("h-isim").textContent = adi;
  document.getElementById("h-no").textContent = no;
  document.getElementById("s-hdr").style.background = hb.renk;
  soruOlustur();
  sureBaslat();
  tabKontrol();
}

function goster(id) {
  document.querySelectorAll(".ekran").forEach(e=>e.classList.remove("aktif"));
  document.getElementById(id).classList.add("aktif");
}

function sureBaslat() {
  baslangic = Date.now();

function prog() {
  let n = 0;
  sorular.forEach((s,i) => {
    if (s.tip==="tanim" && document.getElementById(`c${i}`)?.value?.trim()) n++;
    else if (s.tip==="sayisal" && document.getElementById(`c${i}_${s.alanlar[0].key}`)?.value?.trim()) n++;
    else if (s.tip==="bosluk" && document.getElementById(`c${i}_b0`)?.value?.trim()) n++;
  });

  document.getElementById("p-t").textContent = `${n} / ${TOPLAM_SORU}`;
}

async function bitir() {
  clearInterval(timer);
  goster("yukleme");
  const gecen = Math.floor((Date.now() - baslangic) / 1000);
  const toplanmis = topla();

  let sonuc;
  try {
    sonuc = await claudeDegerlendir(toplanmis);
  } catch(e) {
    sonuc = yerelDegerlendir(toplanmis);
  }

  // Google Sheets'e kaydet
  await sheetsKaydet(sonuc, gecen);
  gosterSonuc(sonuc, gecen);
}

function topla() {
  return sorular.map((s,i) => {
    if (s.tip==="tanim") return {...s, ogCevap: document.getElementById(`c${i}`)?.value?.trim()||""};
    else if (s.tip==="sayisal") {
      const vals={};
      s.alanlar.forEach(a => vals[a.key]=document.getElementById(`c${i}_${a.key}`)?.value?.trim()||"");
      return {...s, vals};
    } else {
      const vals = s.satirlar.map((r,j)=>({...r,ogCevap:document.getElementById(`c${i}_b${j}`)?.value?.trim()||""}));
      return {...s, vals};
    }
  });
}

async function claudeDegerlendir(toplanmis) {
  const tanimlar = toplanmis.filter(s=>s.tip==="tanim");
  const yerel = yerelDegerlendir(toplanmis);

  const prompt = `Haritacılıkta Mesleki Hesaplamalar dersi, MYO 1. sınıf öğrencisi.
Öğrenci: ${adi} (${no}) — Hafta ${HAFTA_NO}

${tanimlar.map((s,i)=>`
[${i+1}] Maks ${s.puan} puan
Soru: ${s.soru}
Öğrenci: "${s.ogCevap}"
Anahtar: ${s.anahtarlar.join(", ")}
İdeal: ${s.ideal}`).join("")}

MYO düzeyinde değerlendir, çok katı olma.
SADECE JSON döndür:
{"d":[${tanimlar.map((s,i)=>`{"i":${i},"p":0,"f":""}`).join(",")}]}
p=puan(0-${Math.max(...tanimlar.map(s=>s.puan))}), f=1 cümle Türkçe geri bildirim`;

  const resp = await fetch("https://api.anthropic.com/v1/messages",{
    method:"POST", headers:{"Content-Type":"application/json"},
    body:JSON.stringify({model:"claude-sonnet-4-20250514",max_tokens:600,
      messages:[{role:"user",content:prompt}]})
  });
  const data = await resp.json();
  const json = JSON.parse(data.content[0].text.replace(/```json|```/g,"").trim());

  const tIdx = toplanmis.map((s,i)=>s.tip==="tanim"?i:-1).filter(i=>i>=0);
  json.d.forEach((d,di) => {
    const s = yerel.sorular[tIdx[di]];
    if (s) { yerel.toplamPuan -= s.alinan; s.alinan=Math.max(0,Math.min(d.p,s.max)); s.geri=d.f; s.aiYapti=true; yerel.toplamPuan+=s.alinan; }
  });
  return yerel;
}

// ═══════════════════════════════════════════════════════════
//  YEREL DEĞERLENDİRME
// ═══════════════════════════════════════════════════════════
function yerelDegerlendir(toplanmis) {
  let toplam = 0;
  const srs = toplanmis.map(s => {
    if (s.tip==="tanim") {
      const c=(s.ogCevap||"").toLowerCase();
      const b=s.anahtarlar.filter(k=>c.includes(k.toLowerCase())).length;
      const alinan=Math.round(s.puan*Math.min(b/s.anahtarlar.length*1.6,1));
      toplam+=alinan;
      return {id:s.id||0,tip:"tanim",konu:s.konu,soru:s.soru,alinan,max:s.puan,
        ogCevap:s.ogCevap,ideal:s.ideal,
        geri:alinan===s.puan?"Mükemmel!":alinan>0?"Kısmi doğru.":"Eksik veya boş."};
    } else if (s.tip==="sayisal") {
      let puan=0;
      const detay=s.alanlar.map(a=>{
        const g=parseFloat((s.vals[a.key]||"").replace(",","."));
        const d=!isNaN(g)&&Math.abs(g-a.dogru)<=a.tol;
        if(d) puan+=s.puan/s.alanlar.length;
        return {label:a.label,dogru:d,girilen:s.vals[a.key]||"—",beklenen:String(a.dogru)};
      });
      puan=Math.round(puan); toplam+=puan;
      return {tip:"sayisal",konu:s.konu,soru:s.soru,alinan:puan,max:s.puan,detay,cozum:s.cozum};
    } else {
      let puan=0;
      const detay=s.vals.map(v=>{
        const g=(v.ogCevap||"").toLowerCase().replace(/\s/g,"");
        const d=v.ak.some(k=>g.includes(k.toLowerCase().replace(/\s/g,"")));
        if(d) puan+=s.puan/s.satirlar.length;
        return {label:v.l,dogru:d,girilen:v.ogCevap||"—",beklenen:v.d};
      });
      puan=Math.round(puan); toplam+=puan;
      return {tip:"bosluk",konu:s.konu,alinan:puan,max:s.puan,detay};
    }
  });
  return {toplamPuan:Math.min(toplam,100),sorular:srs};
}

let sheetsOk = false;
async function sheetsKaydet(veri, gecen) {
  if (SHEETS_URL === "BURAYA_APPS_SCRIPT_URL_YAPISTIR") return;
  try {
    const tam = veri.sorular.filter(s=>s.alinan===s.max).length;
    const yanl = veri.sorular.filter(s=>s.alinan===0).length;
    const kism = veri.sorular.length - tam - yanl;
    await fetch(SHEETS_URL, {
      method:"POST",
      headers:{"Content-Type":"text/plain"},
      body: JSON.stringify({
        hafta: HAFTA_NO, adi, no,
        puan: veri.toplamPuan,
        sure: gecen,
        dogru: tam + kism,
        yanlis: yanl,
        tamDogru: tam,
        kismi: kism,
        seed,
        tabUyari
      })
    });
    sheetsOk = true;
  } catch(e) { sheetsOk = false; }
}
